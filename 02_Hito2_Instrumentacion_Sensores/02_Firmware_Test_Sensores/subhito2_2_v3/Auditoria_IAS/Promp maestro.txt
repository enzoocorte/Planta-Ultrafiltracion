## 🏭 1. Contexto del Proceso Físico y Restricciones Duras
1. **El Proceso:** Planta Piloto de Ultrafiltración de flujo tangencial (*cross-flow*) para una tesis de grado (Universidad Nacional de Salta, Argentina).
   - **Cartucho de Membrana:** Fresenius FX100 (fibras capilares huecas de polisulfona modificada Helixone® de tecnología nanométrica).
     - **Superficie efectiva ($A_m$):** $2.2\text{ m}^2$.
     - **Coeficiente de ultrafiltración ($K_{\text{UF}}$):** $73\text{ mL/(h}\cdot\text{mmHg)}$ (régimen de alto flujo / *high-flux*).
     - **Geometría de fibras:** Diámetro interno de capilar $d_i = 185\ \mu\text{m}$, grosor de pared $\delta = 35\ \mu\text{m}$.
     - **Morfología de poros:** Poros lisos y cilíndricos en la capa interna activa, optimizados para mínima resistencia al transporte convectivo.
   - **Límite Hidráulico Crítico:** Caudal máximo seguro de membrana: **$600\text{ mL/min}$** ($\approx 36\text{ RPM}$ con $13.60\text{ mL/rev}$). Presión transmembrana máxima permitida: **$\text{TMP} \le 0.50\text{ bar}$**. Por encima de estos valores existe riesgo inminente de ruptura de capilares y colapso estructural.
2. **Subsistema de Impulsión:**
   - Motor paso a paso NEMA 34 (4.5 Nm, con corriente ajustada a ~3A).
   - Driver industrial **Leadshine DM860**. Conexión en **Cátodo Común** (PUL- y DIR- a GND común; GPIOs del ESP32 envían HIGH a 3.3V / ~8mA a PUL+ y DIR+ para activar los optoacopladores).
   - Cabezal peristáltico industrial **MBP-2000** con manguera de silicona de $12\text{ mm}$ de diámetro interno.
   - **Restricción Mecánica:** La bomba y el cabezal **NO deben desmontarse ni modificarse mecánicamente**.
   - Desplazamiento real calibrado: $\approx 13.60\text{ mL/vuelta}$ (680 mL/min a 50 RPM).
   - Resolución del driver: **3200 micropasos/rev** (1/16 paso, SW5=OFF, SW6=OFF, SW7=ON, SW8=ON) para eliminar el torque ripple y la resonancia acústica.
3. **Subsistema de Instrumentación (Caudalímetros):**
   - Sensores de turbina con transistor Hall NPN en colector abierto (**YF-S401**):
     - `PIN_SENSOR_ALIMENTACION = 14` (Salida de bomba / Feed).
     - `PIN_SENSOR_PERMEADO = 27` (Permeado filtrado).
   - **Arquitectura de Doble Bornera Shield (ZS-1057):**
     - Placa 1: CPU Master (ESP32 DevKit V1 de 30/38 pines montado).
     - Placa 2: Módulo Front-End pasivo. Contiene resistencias pull-up externas de **$4.7\text{ k}\Omega$ a 3.3V** y capacitores cerámicos pasabajos de **$100\text{ nF}$ a GND** ($f_c \approx 338\text{ Hz}$).
   - Salida del sensor Hall: Polarizada a 3.3V para proteger los pines del ESP32 (los sensores se alimentan con 5V en VCC para garantizar la conmutación del chip Hall).
4. **Estado Actual del Proyecto (Hito 2.2 V3):**
   - Se resolvió un bug donde el sensor marcaba 81 Hz (835.5 mL/min) con bomba detenida: el filtro por software de $12000\ \mu\text{s}$ ($12\text{ ms}$) actuaba como divisor de frecuencia saturándose ante el zumbido de 50/100 Hz de la red. Se redujo el blanking software a $2000\ \mu\text{s}$ ($2\text{ ms}$) y se colocó el filtro RC físico.
   - Se corrigió una inversión física de cables (el sensor de bomba estaba conectado en P27 en vez de P14).
   - Se incorporó persistencia en Flash NVS (`Preferences.h`), rampa cuadrática S-Curve, SoftAP Wi-Fi dedicado (`192.168.4.1`), exportador CSV y datalogger en RAM multi-sesión.

// Bomba.cpp
#include "Bomba.h"
void Bomba::begin() {
  pinMode(PIN_DIR, OUTPUT);
  fijarSentido(true);
#if defined(ESP_ARDUINO_VERSION) && (ESP_ARDUINO_VERSION >= ESP_ARDUINO_VERSION_VAL(3, 0, 0))
  ledcAttach(PIN_PUL, 800, 10);
  ledcWrite(PIN_PUL, 0); // Reposo: LOW -> Opto OFF
#else
  ledcSetup(0, 800, 10);
  ledcAttachPin(PIN_PUL, 0);
  ledcWrite(0, 0);
#endif
}
void Bomba::arrancar() { _enMarcha = true; }
void Bomba::detener()  { _enMarcha = false; _invirtiendo = false; }
void Bomba::setRPM(float rpm) {
  _objetivo = constrain(rpm, RPM_MIN, RPM_MAX);
}
void Bomba::toggleSentido() {
  if (_invirtiendo) return;
  if (_actual < 1.0f) {
    fijarSentido(!_horario);
  } else {
    _invirtiendo = true;
    _rpmGuardada = _objetivo;
    _objetivo = 0.0f;
  }
}
void Bomba::fijarSentido(bool horario) {
  _horario = horario;
  digitalWrite(PIN_DIR, horario ? LOW : HIGH);
}
void Bomba::tick(float dt) {
  float objetivo = _enMarcha ? _objetivo : 0.0f;
  if (_actual < objetivo) {
    if (_actual < 1.0f) _actual = 1.0f;
    float tasaAcel = ACEL_NOMINAL_RPM_S;
    if (_actual < 10.0f) {
      tasaAcel = ACEL_ARRANQUE_RPM_S + (_actual / 10.0f) * (ACEL_NOMINAL_RPM_S - ACEL_ARRANQUE_RPM_S);
    }
    float delta = objetivo - _actual;
    if (delta < 3.0f) tasaAcel = max(0.8f, tasaAcel * (delta / 3.0f));
    _actual = min(objetivo, _actual + tasaAcel * dt);
  } else if (_actual > objetivo) {
    if (!_enMarcha) {
      _actual = max(0.0f, _actual - FRENADO_PARADA_RPM_S * dt);
      if (_actual < 1.0f) _actual = 0.0f;
    } else {
      float tasaDecel = DESACEL_AJUSTE_RPM_S;
      float delta = _actual - objetivo;
      if (delta < 3.0f) tasaDecel = max(1.0f, tasaDecel * (delta / 3.0f));
      _actual = max(objetivo, _actual - tasaDecel * dt);
    }
  }
  if (_invirtiendo && _actual <= 0.1f) {
    fijarSentido(!_horario);
    _invirtiendo = false;
    _objetivo = _rpmGuardada;
  }
  if (_actual >= 1.0f) {
    uint32_t f = (uint32_t)(_actual * (float)_pulsosPorRev / 60.0f);
    if (f < 50) f = 50;
#if defined(ESP_ARDUINO_VERSION) && (ESP_ARDUINO_VERSION >= ESP_ARDUINO_VERSION_VAL(3, 0, 0))
    if (f != _fActual) { ledcChangeFrequency(PIN_PUL, f, 10); _fActual = f; }
    ledcWrite(PIN_PUL, 512); // Duty 50%
#else
    if (f != _fActual) { ledcSetup(0, f, 10); _fActual = f; }
    ledcWrite(0, 512);
#endif
  } else if (_fActual != 0) {
#if defined(ESP_ARDUINO_VERSION) && (ESP_ARDUINO_VERSION >= ESP_ARDUINO_VERSION_VAL(3, 0, 0))
    ledcWrite(PIN_PUL, 0);
#else
    ledcWrite(0, 0);
#endif
    _fActual = 0;
  }
}

private:
  static void IRAM_ATTR isrPuente(void* arg);
  const uint8_t _pin;
  float         _k;
  const char*   _nombre;
  volatile uint32_t _pulsos = 0;
  volatile uint32_t _t_ultimo = 0;
  float _f = 0.0f, _q = 0.0f, _vol = 0.0f;
  uint8_t _segSinPulso = 0;
  bool _fallo = false;
  static portMUX_TYPE _mux;
};

// caudalimetro.cpp
#include "caudalimetro.h"
portMUX_TYPE Caudalimetro::_mux = portMUX_INITIALIZER_UNLOCKED;
Caudalimetro::Caudalimetro(uint8_t pin, float k, const char* nombre)
  : _pin(pin), _k(k), _nombre(nombre) {}
void Caudalimetro::begin() {
  pinMode(_pin, INPUT_PULLUP);
  attachInterruptArg(digitalPinToInterrupt(_pin), isrPuente, this, FALLING);
}
void Caudalimetro::actualizar(float dt_s, bool bombaEmpuja) {
  portENTER_CRITICAL(&_mux);
  uint32_t n = _pulsos;
  _pulsos = 0;
  portEXIT_CRITICAL(&_mux);
  _f = (dt_s > 0.0f) ? ((float)n / dt_s) : 0.0f;
  float q = (_f * 1000.0f) / _k;
  if (q > Q_MAX_FISICO_MLMIN) {
    q = 0.0f;
  }
  // Suavizado EMA
  _q = (n > 0) ? (0.4f * q + 0.6f * _q) : 0.0f;
  _vol += (float)n / (_k * 60.0f);
  if (n > 0) {
    _segSinPulso = 0;
    _fallo = false;
  } else if (bombaEmpuja) {
    if (++_segSinPulso >= 5) _fallo = true;
  } else {
    _segSinPulso = 0;
    _fallo = false;
  }
}
void IRAM_ATTR Caudalimetro::isrPuente(void* arg) {
  Caudalimetro* c = reinterpret_cast<Caudalimetro*>(arg);
  uint32_t t = micros();
  if (t - c->_t_ultimo >= FILTRO_RUIDO_US) {
    portENTER_CRITICAL_ISR(&_mux);
    c->_pulsos++;
    portEXIT_CRITICAL_ISR(&_mux);
    c->_t_ultimo = t;
  }
}
